## Proposed `lean/Colony/Examples/NestedLoopInvariant.lean`

```lean
import Colony.Tactic

namespace Colony.Examples.NestedLoopInvariant

structure State where
  left : Nat
  right : Nat
  deriving DecidableEq, Repr

inductive Command where
  | add (amount : Nat)
  | skip
  deriving DecidableEq, Repr

def step (state : State) (command : Command) : State :=
  match command with
  | .add amount => ⟨state.left + amount, state.right + amount⟩
  | .skip => state

def runBatch (commands : List Command) (state : State) : State :=
  match commands with
  | [] => state
  | command :: rest => runBatch rest (step state command)

def run (batches : List (List Command)) (state : State) : State :=
  match batches with
  | [] => state
  | batch :: rest => run rest (runBatch batch state)

private theorem step_preserves (state : State) (command : Command)
    (h : state.left = state.right) :
    (step state command).left = (step state command).right := by
  cases command <;> simp [step] <;> omega

private theorem runBatch_preserves (commands : List Command) :
    ∀ state : State, state.left = state.right →
      (runBatch commands state).left = (runBatch commands state).right := by
  induction commands with
  | nil =>
      intro state h
      simpa [runBatch] using h
  | cons command rest ih =>
      intro state h
      simp only [runBatch]
      exact ih (step state command) (step_preserves state command h)

theorem run_preserves (batches : List (List Command)) (state : State)
    (h : state.left = state.right) :
    (run batches state).left = (run batches state).right := by
  fail_if_success
    (have : (run batches state).left = (run batches state).right := by
      colony)
  induction batches generalizing state with
  | nil => simpa [run] using h
  | cons batch rest ih =>
      simp only [run]
      exact ih (runBatch batch state) (runBatch_preserves batch state h)

example : run [[.add 2, .skip], [.add 3]] ⟨1, 1⟩ = ⟨6, 6⟩ := by
  decide

example : True := by
  fail_if_success
    (have : (run [[.add 2, .skip], [.add 3]] ⟨1, 1⟩).left =
        (run [[.add 2, .skip], [.add 3]] ⟨1, 1⟩).right + 1 := by
      colony)
  trivial

example : ¬ (run [[.add 2, .skip], [.add 3]] ⟨1, 1⟩).left =
    (run [[.add 2, .skip], [.add 3]] ⟨1, 1⟩).right + 1 := by
  decide

#print axioms Colony.Examples.NestedLoopInvariant.run_preserves

end Colony.Examples.NestedLoopInvariant
```
